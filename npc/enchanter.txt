//===== rAthena Script =======================================
//= Universal Enchanter (Class Stones Enabled)
//===== By: Lil Art ===========================================
//===== Description: =========================================
//= Free -- no Zeny cost, just consumes the stone used. Accepts
//= any inventory item (Class Stones, Garment Stones, Essences,
//= Fixed Casting, Nebula Stones, etc.) and attaches it to an
//= empty socket of equipped Costume, Shadow, or Regular gear.
//= Regular gear: only the sockets after its card slots, so the
//= card slots stay for cards.
//= Offered: enchant-type items, the enchants the item db leaves
//= typed as plain cards (headgear jewels, costume stones -- not
//= monster cards or Tarot cards); only the ones whose item db
//= places include the chosen slot (#46). Etc items are never
//= offered: none has a script, so in a socket they do nothing
//= (Etel Stone, Pink Glow Effect (Middle), ...).
//= The item keeps its refine, grade, random options, bound type
//= and broken state. Refused: rental items, forged/brewed/pet
//= items (their card fields hold crafter data), and items the
//= player owns more than one copy of (equip picks by item ID).
//= Add a slot (#111), only with Lil Art's cash-shop-extended loaded
//= (it sells the coupon): one Slotting Advertisement (6188) turns an
//= item from the official slotting NPCs' list (Seiyablem, Leablem)
//= into its slotted twin, with their success chance (5-25%, Hat of
//= the Sun God 90%). The coupon is used up either way; a failure
//= leaves the item as it was (officially it is destroyed). Keeps
//= the same things and refuses the same items as the enchanting.
//===== How to customize: ====================================
//= - Add a slot list: generated by tools/slot-table.cjs from the
//=   rAthena the app pins (repo only); rerun it, don't edit by hand.
//= - Which slots are offered: the .@EQ_SLOTS/.@EQ_POS/.@EQ_NAMES$
//=   arrays for each category (Costume, Shadow, Regular). .@EQ_POS
//=   is the EQP_* position matching each EQI_* slot.
//= - Location/sprite: the first line below.
//============================================================

prontera,164,169,4	script	Universal Enchanter	10182,{
    mes "[Universal Enchanter]";
    mes "Greetings! I can attach any ^0055FFEnchant Stone or Class Stone^000000 directly to your equipped gear!";
    next;

    // Primary Equipment Category Selection. "Add a slot" only with Lil
    // Art's cash-shop-extended, which sells the Slotting Advertisement;
    // an empty entry is not shown and the numbering stays the same.
    .@slotting$ = "";
    if (getnpcid(0, "CashPointRewards")) .@slotting$ = "Add a slot";
    .@cat = select("Costume Gear", "Shadow Gear", "Regular Gear", .@slotting$, "Cancel");
    if (.@cat == 4) goto S_AddSlot;
    if (.@cat == 5) close;

    deletearray .@EQ_SLOTS[0];
    deletearray .@EQ_POS[0];
    deletearray .@EQ_NAMES$[0];

    // Branch equipment choices by category
    if (.@cat == 1) {
        setarray .@EQ_SLOTS[1], EQI_COSTUME_HEAD_TOP, EQI_COSTUME_HEAD_MID, EQI_COSTUME_HEAD_LOW, EQI_COSTUME_GARMENT;
        setarray .@EQ_POS[1], EQP_COSTUME_HEAD_TOP, EQP_COSTUME_HEAD_MID, EQP_COSTUME_HEAD_LOW, EQP_COSTUME_GARMENT;
        setarray .@EQ_NAMES$[1], "Costume Top Headgear", "Costume Mid Headgear", "Costume Low Headgear", "Costume Garment";
    } else if (.@cat == 2) {
        setarray .@EQ_SLOTS[1], EQI_SHADOW_ARMOR, EQI_SHADOW_WEAPON, EQI_SHADOW_SHIELD, EQI_SHADOW_SHOES, EQI_SHADOW_ACC_R, EQI_SHADOW_ACC_L;
        setarray .@EQ_POS[1], EQP_SHADOW_ARMOR, EQP_SHADOW_WEAPON, EQP_SHADOW_SHIELD, EQP_SHADOW_SHOES, EQP_SHADOW_ACC_R, EQP_SHADOW_ACC_L;
        setarray .@EQ_NAMES$[1], "Shadow Armor", "Shadow Weapon", "Shadow Shield", "Shadow Shoes", "Shadow Accessory (Right)", "Shadow Accessory (Left)";
    } else if (.@cat == 3) {
        setarray .@EQ_SLOTS[1], EQI_HEAD_TOP, EQI_HEAD_MID, EQI_HEAD_LOW, EQI_ARMOR, EQI_HAND_R, EQI_HAND_L, EQI_GARMENT, EQI_SHOES, EQI_ACC_R, EQI_ACC_L;
        setarray .@EQ_POS[1], EQP_HEAD_TOP, EQP_HEAD_MID, EQP_HEAD_LOW, EQP_ARMOR, EQP_HAND_R, EQP_HAND_L, EQP_GARMENT, EQP_SHOES, EQP_ACC_R, EQP_ACC_L;
        setarray .@EQ_NAMES$[1], "Top Headgear", "Mid Headgear", "Low Headgear", "Armor", "Right Hand (Weapon)", "Left Hand (Shield)", "Garment", "Shoes", "Accessory (Right)", "Accessory (Left)";
    }

    // Step 1: Select Slot
    mes "[Universal Enchanter]";
    mes "Which equipped piece do you want to enchant?";
    next;

    .@menu_str$ = "";
    for (.@i = 1; .@i <= getarraysize(.@EQ_SLOTS); .@i++) {
        .@eq_id = getequipid(.@EQ_SLOTS[.@i]);
        if (.@eq_id > 0) {
            .@menu_str$ += .@EQ_NAMES$[.@i] + " [" + getitemname(.@eq_id) + "]";
        } else {
            .@menu_str$ += .@EQ_NAMES$[.@i] + " [Empty]";
        }
        .@menu_str$ += ":";
    }
    .@menu_str$ += "Cancel";

    .@sel_slot = select(.@menu_str$);
    if (.@sel_slot > getarraysize(.@EQ_SLOTS)) close;

    .@target_eq = .@EQ_SLOTS[.@sel_slot];
    .@target_id = getequipid(.@target_eq);

    if (.@target_id <= 0) {
        mes "[Universal Enchanter]";
        mes "You do not have gear equipped in that slot!";
        close;
    }

    // Read everything the item carries, so it can be recreated as it is.
    // Find the equipped copy by position; count the other copies.
    getinventorylist;
    .@inv = -1;
    .@copies = 0;
    for (.@i = 0; .@i < @inventorylist_count; .@i++) {
        if (@inventorylist_id[.@i] != .@target_id) continue;
        .@copies++;
        if (.@inv < 0 && (@inventorylist_equip[.@i] & .@EQ_POS[.@sel_slot]))
            .@inv = .@i;
    }
    if (.@inv < 0) {
        mes "[Universal Enchanter]";
        mes "I could not find that item in your inventory.";
        close;
    }
    if (.@copies > 1) {
        mes "[Universal Enchanter]";
        mes "You own more than one ^0055FF" + getitemname(.@target_id) + "^000000.";
        mes "Please put the other copies in storage first, so the right one gets enchanted.";
        close;
    }
    if (@inventorylist_expire[.@inv] > 0) {
        mes "[Universal Enchanter]";
        mes "Rental items cannot be enchanted.";
        close;
    }
    .@c0 = @inventorylist_card1[.@inv];
    if (.@c0 == CARD0_FORGE || .@c0 == CARD0_CREATE || .@c0 == CARD0_PET) {
        mes "[Universal Enchanter]";
        mes "Crafted or named items cannot be enchanted: their sockets hold the crafter's data.";
        close;
    }
    .@c1 = @inventorylist_card2[.@inv];
    .@c2 = @inventorylist_card3[.@inv];
    .@c3 = @inventorylist_card4[.@inv];
    .@refine = @inventorylist_refine[.@inv];
    .@attr = @inventorylist_attribute[.@inv];
    .@bound = @inventorylist_bound[.@inv];
    .@grade = @inventorylist_enchantgrade[.@inv];
    for (.@o = 1; .@o <= 5; .@o++) {
        .@opt_id[.@o - 1] = getd("@inventorylist_option_id" + .@o + "[" + .@inv + "]");
        .@opt_val[.@o - 1] = getd("@inventorylist_option_value" + .@o + "[" + .@inv + "]");
        .@opt_par[.@o - 1] = getd("@inventorylist_option_parameter" + .@o + "[" + .@inv + "]");
    }

    // Regular gear: the item's own card slots stay for cards
    .@first = (.@cat == 3) ? getitemslots(.@target_id) : 0;
    if (.@first >= 4) {
        mes "[Universal Enchanter]";
        mes "All sockets of this item are card slots. There is no room for an enchant.";
        close;
    }

    // Step 2: Select Target Socket Slot (0 to 3)
    mes "[Universal Enchanter]";
    mes "Selected: ^0055FF" + getitemname(.@target_id) + "^000000";
    mes "Which socket do you want to enchant?";
    next;

    // Card slots of regular gear are left out: an empty menu entry is
    // not shown, and the numbering of the others stays the same.
    setarray .@sock[0], .@c0, .@c1, .@c2, .@c3;
    .@sock_str$ = "";
    for (.@i = 0; .@i < 4; .@i++) {
        if (.@i >= .@first)
            .@sock_str$ += "Socket " + (.@i + 1) + " [" + (.@sock[.@i] ? getitemname(.@sock[.@i]) : "^00FF00Empty^000000") + "]";
        .@sock_str$ += ":";
    }
    .@sock_str$ += "Cancel";

    .@sock_choice = select(.@sock_str$) - 1;
    if (.@sock_choice >= 4) close;

    if (.@sock[.@sock_choice] > 0) {
        mes "[Universal Enchanter]";
        mes "That socket is already occupied!";
        close;
    }

    // Step 3: Complete Divine Pride Class Stone & Enchantment Scanner
    getinventorylist;
    .@stone_count = 0;
    deletearray .@valid_stone_id[0];

    for (.@i = 0; .@i < @inventorylist_count; .@i++) {
        .@id = @inventorylist_id[.@i];
        .@type = getiteminfo(.@id, ITEMINFO_TYPE);
        .@subtype = getiteminfo(.@id, ITEMINFO_SUBTYPE);
        .@name$ = getitemname(.@id);

        // Filter out all standard gear, consumables, ammo, and quest items
        if (.@type == IT_WEAPON || .@type == IT_ARMOR || .@type == IT_HEALING || .@type == IT_USABLE || .@type == IT_AMMO) {
            continue;
        }

        // Strict Exclusion Rules:
        // 1. Exclude regular monster cards (CARD_NORMAL / 0)
        // 2. Exclude non-enchant items containing "Ticket", "Rough", or elemental materials
        if (strpos(.@name$, "Ticket") != -1 || strpos(.@name$, "Rough") != -1) {
            continue;
        }

        // A card-type item without the Enchant subtype counts as a monster
        // card to the server. Many enchants are like that (the headgear
        // jewels 27421-27426, the costume stones such as ATK Stone (Upper)),
        // so it is taken as an enchant when its name does not end in "Card"
        // -- monster cards are all named "... Card" -- and is not a Tarot
        // card ("The Justice", "Wheel Of Fortune Reversed", ...) (#46).
        .@len = getstrlen(.@name$);
        .@tail$ = "";
        if (.@len >= 4) .@tail$ = substr(.@name$, .@len - 4, .@len - 1);	// not a ternary: it runs both sides
        .@unmarked = 0;
        if (.@type == IT_CARD && .@subtype == CARD_NORMAL && .@tail$ != "Card" && .@tail$ != "card") {
            .@unmarked = 1;
            if (strpos(.@name$, "The ") == 0 || strpos(.@name$, "Wheel Of Fortune") == 0) .@unmarked = 0;
        }

        // Acceptance Criteria:
        // - Item of type IT_CARD that isn't a normal monster card (Nebula, Trait, Essences)
        // - An enchant the item db leaves without the Enchant subtype (above)
        // Etc items are not taken: none has a script in the item db (the
        // 629 Etc "... Stone"/"... Essence" ones included), so in a socket
        // they do nothing and the stone is lost. The working enchants are
        // card-type, e.g. Pink Glow Effect 29161, not the Etc 25178.
        if (!((.@type == IT_CARD && .@subtype != CARD_NORMAL) || .@unmarked)) {
            continue;
        }

        // Only what fits the chosen slot: an item whose db entry names its
        // places must name this one (the jewels: headgear only). No places
        // given (most stones) = any slot, as before.
        .@where = getiteminfo(.@id, ITEMINFO_LOCATIONS);
        if (.@where > 0 && !(.@where & .@EQ_POS[.@sel_slot])) continue;

        .@valid_stone_id[.@stone_count] = .@id;
        .@stone_count++;
    }

    if (.@stone_count == 0) {
        mes "[Universal Enchanter]";
        mes "You do not have any valid Class Stones or Enchant Stones in your inventory!";
        close;
    }

    // Step 4: Choose Stone
    mes "[Universal Enchanter]";
    mes "Select the Class Stone or Enchantment to attach:";
    next;

    .@stone_menu$ = "";
    for (.@i = 0; .@i < .@stone_count; .@i++) {
        .@stone_menu$ += getitemname(.@valid_stone_id[.@i]) + ":";
    }
    .@stone_menu$ += "Cancel";

    .@stone_sel = select(.@stone_menu$) - 1;
    if (.@stone_sel >= .@stone_count) close;

    .@chosen_stone_id = .@valid_stone_id[.@stone_sel];

    mes "[Universal Enchanter]";
    mes "Attach ^0055FF" + getitemname(.@chosen_stone_id) + "^000000 to ^0055FF" + getitemname(.@target_id) + "^000000?";
    next;

    if (select("Yes, proceed:Cancel") == 2) close;

    if (countitem(.@chosen_stone_id) < 1) {
        mes "[Universal Enchanter]";
        mes "You are missing the required stone!";
        close;
    }

    // Recreate the item with everything read above, plus the stone
    .@sock[.@sock_choice] = .@chosen_stone_id;

    delitem .@chosen_stone_id, 1;
    delequip .@target_eq;
    getitembound4 .@target_id, 1, 1, .@refine, .@attr, .@sock[0], .@sock[1], .@sock[2], .@sock[3], .@bound, .@grade, .@opt_id, .@opt_val, .@opt_par;
    equip .@target_id;

    specialeffect2 154;
    mes "[Universal Enchanter]";
    mes "Enchantment successful!";
    close;

// Add a slot (#111): one Slotting Advertisement turns an item from the
// official slotting NPCs' list (Seiyablem, Leablem) into its slotted
// twin, with their success chance. A failure only uses up the coupon.
S_AddSlot:
    // The listed items in the inventory (unequipped or equipped)
    getinventorylist;
    .@n = 0;
    .@menu$ = "";
    // One pass, looked up by item ID: a loop over the table for each
    // item ran into rAthena's "infinity loop" limit with a full bag.
    for (.@i = 0; .@i < @inventorylist_count; .@i++) {
        .@id = @inventorylist_id[.@i];
        .@t = .slot_row[.@id] - 1;
        if (.@t < 0) continue;
        // an item in the bag twice (two copies) is shown once
        if (.@shown[.@id]) continue;
        .@shown[.@id] = 1;
        .@pick_id[.@n] = .@id;
        .@pick_row[.@n] = .@t;
        .@menu$ += getitemname(.@id) + " (" + .slot_chance[.@t] + "%):";
        .@n++;
    }

    mes "[Universal Enchanter]";
    mes "With a ^0055FFSlotting Advertisement^000000 I can add a slot to the equipment the slotting engineers Seiyablem and Leablem work on.";
    mes "If it fails, your item stays as it is -- only the coupon is used up.";
    if (.@n == 0) {
        mes "But you carry none of that equipment.";
        close;
    }
    next;
    mes "[Universal Enchanter]";
    mes "Which item? The success chance is in brackets.";
    next;
    .@sel = select(.@menu$ + "Cancel") - 1;
    if (.@sel >= .@n) close;
    .@from = .@pick_id[.@sel];
    .@to = .slot_to[.@pick_row[.@sel]];
    .@chance = .slot_chance[.@pick_row[.@sel]];

    mes "[Universal Enchanter]";
    mes "^0055FF" + getitemname(.@from) + "^000000 becomes ^0055FF" + getitemname(.@to) + "^000000 [" + getitemslots(.@to) + "].";
    mes "Success chance: ^FF0000" + .@chance + "%^000000. It costs 1 Slotting Advertisement, whatever the result.";
    next;
    if (select("Add a slot:Cancel") == 2) close;

    if (countitem(6188) < 1) {	// Slot_Coupon
        mes "[Universal Enchanter]";
        mes "You need a ^0055FFSlotting Advertisement^000000. The Cash Shop sells them.";
        close;
    }

    // Read the item now, as it is at this moment, so it can be recreated
    getinventorylist;
    .@inv = -1;
    .@copies = 0;
    for (.@i = 0; .@i < @inventorylist_count; .@i++) {
        if (@inventorylist_id[.@i] != .@from) continue;
        .@copies++;
        .@inv = .@i;
    }
    if (.@inv < 0) {
        mes "[Universal Enchanter]";
        mes "I could not find that item in your inventory.";
        close;
    }
    if (.@copies > 1) {
        mes "[Universal Enchanter]";
        mes "You own more than one ^0055FF" + getitemname(.@from) + "^000000.";
        mes "Please put the other copies in storage first, so the right one gets the slot.";
        close;
    }
    if (@inventorylist_expire[.@inv] > 0) {
        mes "[Universal Enchanter]";
        mes "Rental items cannot get a slot.";
        close;
    }
    if (!@inventorylist_identify[.@inv]) {
        mes "[Universal Enchanter]";
        mes "Please have the item appraised first.";
        close;
    }
    .@c0 = @inventorylist_card1[.@inv];
    if (.@c0 == CARD0_FORGE || .@c0 == CARD0_CREATE || .@c0 == CARD0_PET) {
        mes "[Universal Enchanter]";
        mes "Crafted or named items cannot get a slot: their sockets hold the crafter's data.";
        close;
    }
    .@c1 = @inventorylist_card2[.@inv];
    .@c2 = @inventorylist_card3[.@inv];
    .@c3 = @inventorylist_card4[.@inv];
    .@refine = @inventorylist_refine[.@inv];
    .@attr = @inventorylist_attribute[.@inv];
    .@bound = @inventorylist_bound[.@inv];
    .@grade = @inventorylist_enchantgrade[.@inv];
    .@worn = @inventorylist_equip[.@inv];
    for (.@o = 1; .@o <= 5; .@o++) {
        .@opt_id[.@o - 1] = getd("@inventorylist_option_id" + .@o + "[" + .@inv + "]");
        .@opt_val[.@o - 1] = getd("@inventorylist_option_value" + .@o + "[" + .@inv + "]");
        .@opt_par[.@o - 1] = getd("@inventorylist_option_parameter" + .@o + "[" + .@inv + "]");
    }

    delitem 6188, 1;	// Slot_Coupon, Slotting Advertisement
    if (rand(1, 100) > .@chance) {
        specialeffect EF_SUI_EXPLOSION;
        mes "[Universal Enchanter]";
        mes "Ah... it did not take. Your ^0055FF" + getitemname(.@from) + "^000000 is unharmed, but the coupon is gone.";
        close;
    }

    // The one copy goes (worn or not) and comes back slotted, with its
    // cards and enchants in the same sockets
    delitem .@from, 1;
    getitembound4 .@to, 1, 1, .@refine, .@attr, .@c0, .@c1, .@c2, .@c3, .@bound, .@grade, .@opt_id, .@opt_val, .@opt_par;
    if (.@worn) equip .@to;

    specialeffect EF_SANCTUARY;
    mes "[Universal Enchanter]";
    mes "Success! Here is your ^0055FF" + getitemname(.@to) + "^000000.";
    close;
