//===== rAthena Script =======================================
//= Stylist
//===== By: Euphy, MondoTruth ================================
//===== Description: =========================================
//= Opens the styling window (openstylist) for hair style, hair
//= color, cloth color and (Renewal) outfit; the player pays
//= there on Buy, in Zeny as the settings say or with a coupon
//= (New Style Coupon for hair style, Clothing Dye Coupon for
//= cloth color, Costume Change Ticket for outfit; hair color is Zeny
//= only), chosen in the window. A change of several things at
//= once is paid in full or not made at all.
//= Outfit: 3rd classes (also transcendent, mounted and baby)
//= and 4th classes can switch to the 3rd class's alternative
//= outfit (the "second costume"); a 4th class gets its 3rd
//= class's one. Kept on log-in, taken off by a job change.
//===== How to customize: ====================================
//= - Location/sprite: the first line (map,x,y,dir / sprite).
//= - Prices: the Stylist's "Prices..." page in the mod's
//=   settings window (stylist_price_*). Coupons: the S_Cost
//=   lines in OnInit (item ID, 0 = Zeny only).
//=   They need the fork's stylist_script_costs extension, which
//=   db/extension_db.yml switches on. The window learns the
//=   prices and sends the player's way to pay through
//=   "@@event stylist" and @snpcstylist (app docs/MODDING.md).
//= - Which styles the window offers: rAthena's db/re/stylist.yml
//=   (Pre-renewal: this mod's pre-renewal/db/stylist.yml).
//= - Which classes see the outfit tab: the S_Map lines in OnInit.
//=   The server's own list (db/re/job_outfits.yml) decides what
//=   it sells; keep the two the same.
//= License: GPL-3.0, based on rAthena's npc/custom/stylist.txt
//=   (see LICENSE and LICENSE-GPL-3.0).
//============================================================

prontera,169,180,5	script	Stylist#custom_stylist	122,{
    // The outfit only for classes that have one (Renewal only).
    set .@outfit, .outfit[Class] && .outfit$ != "";
    mes "[Stylist]";
    mes "Pick a new look in my styling window. You pay when you press Buy:";
    mes "Hair style: " + .cost$[LOOK_HAIR];
    mes "Hair color: " + .cost$[LOOK_HAIR_COLOR];
    mes "Cloth color: " + .cost$[LOOK_CLOTHES_COLOR];
    if (.@outfit) mes "Outfit: " + .cost$[LOOK_BODY2];
    close2;
    // The prices and the way to pay, for the window (the app's stylist
    // window keeps this line out of chat). The outfit tab only shows when
    // the outfit is in the offer.
    set .@offer$, .offer$;
    if (.@outfit) set .@offer$, .@offer$ + .outfit$;
    if (.@offer$ != "") dispbottom "@@event stylist pay=snpcstylist" + .@offer$;
    openstylist;
    end;

// The window's "Pay with Zeny / coupons": @snpcstylist <request> pay <0|1>,
// sent by the window before the purchase. setstylistpayment needs the
// stylist_script_costs extension; opening the window resets it to Zeny.
OnPay:
    if (.@atcmd_parameters$[1] == "pay" && setstylistpayment(atoi(.@atcmd_parameters$[2])))
        dispbottom "@@reply " + .@atcmd_parameters$[0] + " 1/1 ok";
    else
        dispbottom "@@reply " + .@atcmd_parameters$[0] + " 1/1 refused";
    end;

// Sets one look's cost in the styling window and the text shown for it;
// returns its part of the window's offer (empty if refused).
// arg 0 = the look, 1 = Zeny, 2 = the coupon that pays instead (0 = none),
// 3 = the look's name in the offer.
S_Cost:
    set .@zeny, getarg(1);
    set .@item, getarg(2);
    // A coupon the era's item table does not have: Zeny only.
    if (.@item && getitemname(.@item) == "null") set .@item, 0;
    set .@part$, "";
    if (!setstylistcost(getarg(0), .@zeny, .@item)) {
        // Refused (the reason is in the map server log): stylist.yml's cost stays.
        set .cost$[getarg(0)], "the server's own price";
        return .@part$;
    }
    if (.@zeny) set .@text$, callfunc("F_InsertComma", .@zeny) + " Zeny";
    else set .@text$, "free";
    if (.@item) set .@text$, .@text$ + " or 1 " + getitemname(.@item);
    set .cost$[getarg(0)], .@text$;
    return " " + getarg(3) + "=" + .@zeny + ":" + .@item;

// .outfit[<class>] = the outfit's job ID; arg 0 is the outfit, the rest
// are the classes that may wear it (db/re/job_outfits.yml in the fork).
S_Map:
    for (set .@i, 1; .@i < getargcount(); set .@i, .@i + 1)
        set .outfit[getarg(.@i)], getarg(0);
    return;

OnInit:
    // Switched off in the settings window? Hide.
    if (callfunc("F_SNPC_Off", "stylist")) disablenpc strnpcinfo(0);
    bindatcmd "snpcstylist", strnpcinfo(3) + "::OnPay", 0, 99;
    // The styling window's costs, also for @stylist while the NPC is hidden.
    // Coupons: 7622 New_Style_Coupon, 6046 Clothing_Dye_Coupon.
    set .offer$, callsub(S_Cost, LOOK_HAIR, callfunc("F_ModSetting", "standart-npc", "stylist_price_hair", 100000), 7622, "head");
    set .offer$, .offer$ + callsub(S_Cost, LOOK_HAIR_COLOR, callfunc("F_ModSetting", "standart-npc", "stylist_price_hair_color", 100000), 0, "headpalette");
    set .offer$, .offer$ + callsub(S_Cost, LOOK_CLOTHES_COLOR, callfunc("F_ModSetting", "standart-npc", "stylist_price_cloth_color", 100000), 6046, "bodypalette");
    // Alternative outfits are Renewal only (checkre(0): 1 = Renewal).
    set .outfit$, "";
    if (!checkre(0)) end;
    // Coupon: 6959 Costume_Ticket.
    set .outfit$, callsub(S_Cost, LOOK_BODY2, callfunc("F_ModSetting", "standart-npc", "stylist_price_outfit", 100000), 6959, "bodystyle");
    callsub S_Map, JOB_RUNE_KNIGHT_2ND, JOB_RUNE_KNIGHT, JOB_RUNE_KNIGHT2, JOB_RUNE_KNIGHT_T, JOB_RUNE_KNIGHT_T2, JOB_BABY_RUNE_KNIGHT, JOB_BABY_RUNE_KNIGHT2, JOB_DRAGON_KNIGHT, JOB_DRAGON_KNIGHT2;
    callsub S_Map, JOB_WARLOCK_2ND, JOB_WARLOCK, JOB_WARLOCK_T, JOB_BABY_WARLOCK, JOB_ARCH_MAGE;
    callsub S_Map, JOB_RANGER_2ND, JOB_RANGER, JOB_RANGER2, JOB_RANGER_T, JOB_RANGER_T2, JOB_BABY_RANGER, JOB_BABY_RANGER2, JOB_WINDHAWK, JOB_WINDHAWK2;
    callsub S_Map, JOB_ARCHBISHOP_2ND, JOB_ARCH_BISHOP, JOB_ARCH_BISHOP_T, JOB_BABY_ARCH_BISHOP, JOB_CARDINAL;
    callsub S_Map, JOB_MECHANIC_2ND, JOB_MECHANIC, JOB_MECHANIC2, JOB_MECHANIC_T, JOB_MECHANIC_T2, JOB_BABY_MECHANIC, JOB_BABY_MECHANIC2, JOB_MEISTER, JOB_MEISTER2;
    callsub S_Map, JOB_GUILLOTINE_CROSS_2ND, JOB_GUILLOTINE_CROSS, JOB_GUILLOTINE_CROSS_T, JOB_BABY_GUILLOTINE_CROSS, JOB_SHADOW_CROSS;
    callsub S_Map, JOB_ROYAL_GUARD_2ND, JOB_ROYAL_GUARD, JOB_ROYAL_GUARD2, JOB_ROYAL_GUARD_T, JOB_ROYAL_GUARD_T2, JOB_BABY_ROYAL_GUARD, JOB_BABY_ROYAL_GUARD2, JOB_IMPERIAL_GUARD, JOB_IMPERIAL_GUARD2;
    callsub S_Map, JOB_SORCERER_2ND, JOB_SORCERER, JOB_SORCERER_T, JOB_BABY_SORCERER, JOB_ELEMENTAL_MASTER;
    callsub S_Map, JOB_MINSTREL_2ND, JOB_MINSTREL, JOB_MINSTREL_T, JOB_BABY_MINSTREL, JOB_TROUBADOUR;
    callsub S_Map, JOB_WANDERER_2ND, JOB_WANDERER, JOB_WANDERER_T, JOB_BABY_WANDERER, JOB_TROUVERE;
    callsub S_Map, JOB_SURA_2ND, JOB_SURA, JOB_SURA_T, JOB_BABY_SURA, JOB_INQUISITOR;
    callsub S_Map, JOB_GENETIC_2ND, JOB_GENETIC, JOB_GENETIC_T, JOB_BABY_GENETIC, JOB_BIOLO;
    callsub S_Map, JOB_SHADOW_CHASER_2ND, JOB_SHADOW_CHASER, JOB_SHADOW_CHASER_T, JOB_BABY_SHADOW_CHASER, JOB_ABYSS_CHASER;
    end;
}
